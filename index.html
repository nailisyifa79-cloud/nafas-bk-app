async function activateLicense() { const name = document.getElementById('licenseName').value.trim(); const school = document.getElementById('licenseSchool').value.trim(); const code = document.getElementById('licenseCode').value.trim().toUpperCase(); const errorEl = document.getElementById('licenseError'); const successEl = document.getElementById('licenseSuccess'); errorEl.classList.remove('show'); successEl.classList.remove('show'); if (!name || !school || !code) { errorEl.textContent = '⚠️ Semua field harus diisi.'; errorEl.classList.add('show'); return; } // Validasi format kode (NFS-XXXX-XXXX) if (!/^NFS-[A-Z0-9]{4}-[A-Z0-9]{4}$/.test(code) && !code.startsWith('TEST-')) { errorEl.textContent = '⚠️ Format kode tidak valid. Contoh: NFS-ABCD-1234'; errorEl.classList.add('show'); return; } try { // 1. Cek validasi kode const validateResponse = await fetch(LICENSE_API_URL, { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ action: 'validateLicense', license: code }) }); const validateResult = await validateResponse.json(); if (validateResult.status !== 'OK') { errorEl.textContent = validateResult.message; errorEl.classList.add('show'); return; } // 2. Aktivasi const activateResponse = await fetch(LICENSE_API_URL, { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ action: 'activateLicense', license: code, nama: name, sekolah: school, email: '' // Bisa tambahkan email jika ada di form }) }); const activateResult = await activateResponse.json(); if (activateResult.status === 'OK') { const licenseInfo = { name: name, school: school, code: code, activated: new Date().toISOString(), expiry: new Date(Date.now() + 365 * 24 * 60 * 60 * 1000).toISOString() }; localStorage.setItem('NAFAS_LICENSE_DATA', JSON.stringify(licenseInfo)); localStorage.setItem('NAFAS_LICENSE_USED', code); successEl.textContent = activateResult.message; successEl.classList.add('show'); setTimeout(() => enterApp(licenseInfo), 1000); } else { errorEl.textContent = activateResult.message; errorEl.classList.add('show'); } } catch(e) { errorEl.textContent = '❌ Gagal terhubung ke server lisensi. Periksa koneksi internet.'; errorEl.classList.add('show'); console.error('Error:', e); } }
